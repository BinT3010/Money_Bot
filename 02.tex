\section{So sánh Basic Virtual Assistant và Smart Virtual Assistant}

\subsection*{Mô tả khái quát}
\begin{itemize}
    \item \textbf{Basic Virtual Assistant (BVA):} Hoạt động thụ động dựa trên các quy tắc tĩnh (Rule-based), yêu cầu người dùng phải tự nhập liệu hoặc thao tác thủ công; chỉ thực hiện các phép tính số học đơn giản và đưa ra thông báo cố định khi được yêu cầu.
    \item \textbf{Smart Virtual Assistant (SVA):} Tận dụng AI, xử lý ngôn ngữ tự nhiên (NLP) và phân tích dữ liệu lịch sử để tự động hóa; có khả năng học hỏi thói quen, chủ động đề xuất hạn mức thông minh, giám sát thời gian thực và đưa ra các khuyến nghị hành động ngữ cảnh cá nhân hóa.
\end{itemize}

\textbf{Thuộc Mục 1: Tổng hợp \& Trực quan hóa}
\begin{itemize}
    \item Chỉ hỗ trợ người dùng nhập liệu thủ công bằng tay từng trường dữ liệu (số tiền, nội dung chi tiêu).
    \item Hoàn toàn không có khả năng quét hóa đơn bằng thuật toán OCR hay nhận diện giọng nói.
    \item Người dùng phải tự chọn danh mục chi tiêu cho từng giao dịch từ danh sách thả xuống.
    \item Cộng tổng số tiền theo từng danh mục bằng các công thức số học cố định.
    \item Vẽ biểu đồ hoặc hiển thị bảng thống kê một cách thụ động (chỉ hiển thị khi bấm nút yêu cầu).
\end{itemize}
& 
\textbf{Thuộc Mục 1: Tổng hợp \& Trực quan hóa}
\begin{itemize}
    \item Tiếp nhận dữ liệu đa phương thức thông minh: nhập văn bản tự nhiên, sử dụng OCR quét ảnh/PDF hóa đơn và nhận diện giọng nói (Speech-to-Text).
    \item AI tự động trích xuất các thực thể chính (Số tiền, Ngày, Đơn vị/Cửa hàng, Nội dung chi tiết).
    \item Tự động phân loại giao dịch vào 6 danh mục chuẩn dựa trên ngữ cảnh kèm chỉ số tin cậy (Yêu cầu xác nhận nếu độ tin cậy $< 70\%$).
    \item Tự động tổng hợp và trực quan hóa cơ cấu chi tiêu đa chiều ngay khi có giao dịch mới.
    \item Nhận diện mẫu hành vi tiêu dùng và phát hiện các biến động bất thường so với tháng trước.
\end{itemize} \\
\hline

\textbf{Thuộc Mục 2: Đề xuất hạn mức tự động}
\begin{itemize}
    \item Người dùng phải tự tính toán và nhập hạn mức cứng cho từng danh mục bằng tay.
    \item Hoặc áp dụng công thức chia tỷ lệ cố định, cứng nhắc (chia đều các mục) không xét hoàn cảnh thực tế.
    \item Hạn mức giữ nguyên qua các tháng trừ khi người dùng chủ động điều chỉnh thủ công.
\end{itemize}
& 
\textbf{Thuộc Mục 2: Đề xuất hạn mức tự động}
\begin{itemize}
    \item Tự động phân tích lịch sử chi tiêu ít nhất 3 tháng kết hợp thu nhập đầu vào (7.000.000 đ) để nhận diện xu hướng.
    \item Tự động tính toán và gợi ý hạn mức (\textit{budget ceiling}) tối ưu cho từng danh mục sát với thực tế sinh hoạt sinh viên.
    \item Linh hoạt gợi ý điều chỉnh hạn mức theo mùa vụ (ví dụ: tháng thi cử tăng tiền in ấn, đầu kỳ tăng tiền sách vở).
\end{itemize} \\
\hline

\textbf{Thuộc Mục 3: Giám sát real-time \& Cảnh báo}
\begin{itemize}
    \item Chỉ kiểm tra ngân sách khi người dùng chủ động mở ứng dụng xem số dư còn lại.
    \item Không có cơ chế cảnh báo đa tầng; chỉ thông báo khi số tiền chạm mốc 0 đ hoặc âm tiền.
    \item Thông báo dạng hệ thống khô khan: \textit{"Bạn đã tiêu X đồng, còn lại Y đồng"}.
    \item Không thể đưa ra lời khuyên hay phương án khắc phục bội chi.
\end{itemize}
& 
\textbf{Thuộc Mục 3: Giám sát real-time \& Cảnh báo}
\begin{itemize}
    \item Giám sát liên tục theo thời gian thực (real-time) ngay khi giao dịch mới phát sinh.
    \item Tự động kích hoạt cảnh báo đa tầng: Cảnh báo nhắc nhở ở mốc 80\% và Báo động đỏ khi vượt 100\% (bội chi).
    \item Đưa ra thông báo ngữ cảnh tự nhiên chỉ rõ: tiêu vào việc gì, tại sao chạm ngưỡng.
    \item Lời khuyên hành động thông minh: khuyên giảm chi tiêu cụ thể, chi thêm hợp lý và gợi ý tự động trích quỹ dự phòng bù trừ thâm hụt.
\end{itemize} \\
